    await ocrWorker.setParameters({tessedit_char_whitelist:'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789.-_ '});

    ocrLoading=false;lm.style.display='none';return ocrWorker;

  }catch(e){ocrLoading=false;lm.style.display='none';return null}

}

async function processOcrFrame(){

  if(!ocrWorker||st.activeTab!=='texto')return null;

  const v=document.getElementById('camera-video'),c=document.getElementById('qr-canvas'),ctx=c.getContext('2d',{willReadFrequently:true});

  if(v.readyState!==v.HAVE_ENOUGH_DATA)return null;

  const tw=Math.min(800,v.videoWidth),r=tw/v.videoWidth;c.width=tw;c.height=v.videoHeight*r;ctx.drawImage(v,0,0,c.width,c.height);

  try{const res=await ocrWorker.recognize(c),text=(res.data.text||'').replace(/\s+/g,' ');const cm=text.match(PATTERN_CBU);if(cm&&isValidCBU(cm[1]))return{type:'cbu',value:cm[1],bank:getBankFromCBU(cm[1])};const am=text.match(PATTERN_ALIAS);if(am){const al=am[1].toLowerCase();if(al.length>=6&&al.length<=30&&!al.includes('..'))return{type:'alias',value:al,bank:'Cuenta bancaria'}}return null}catch(e){return null}

}
